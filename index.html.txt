        function checkLogin() {
            if(document.getElementById('username').value === "admin" && document.getElementById('password').value === "p2t") {
                document.getElementById('login-section').style.display = 'none';
                document.getElementById('dashboard').style.display = 'grid';
                load(0, document.querySelector('.nav-btn'));
            } else { alert("Invalid Credentials!"); }
        }

        function load(i, btn) {
            document.getElementById('frame').style.display = 'block';
            document.getElementById('employee-view').style.display = 'none';
            document.getElementById('frame').src = sheets[i];
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
        }

        function loadEmployeePortal(btn) {
            document.getElementById('frame').style.display = 'none';
            document.getElementById('employee-view').style.display = 'block';
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
        }

        setInterval(() => {
            document.getElementById('clock').innerText = new Date().toLocaleTimeString();
        }, 1000);
